//Java class methods assignment part A

class Main {
    public static void main(String[] args) {
        showWelcome();
         printSum();
         displayNumber(1);
         displayNumber(2);
         displayNumber(3);
         int result = multiply(2,3);
         System.out.println(result);
         isEven(2);
        printTable(3);
        System.out.println(sumFromOneToN(5));
        System.out.println(sumFromOneToN(10));
        countDigits(7);
        countDigits(204);
        countDigits(98765);
    }


static void showWelcome()
    {
       System.out.println("Welcome to JAVA");
    }

static void printSum()
    {
        System.out.println(15 + 27);
    }
static void displayNumber(int n)
    {
        System.out.println(n);
    
    }
static int multiply(int a, int b)
    {
        int product = a * b;
        return product;
    }
static void isEven(int number) 
    {
        if (number % 2 == 0)
        {
            System.out.println("True");
        }
         else
        {
            System.out.println("False");
        }
        
    }
static void printTable(int number) 
    {
        for (int num = 1;num <=10;num++)
        {
            int res = number * num;
             System.out.println(number + "X" + num + "=" + res);
        }
        
    }
static int sumFromOneToN(int n) 
    {
        int sum = 0;
        for (int i =1;i<=n;i++)
            {
                sum = sum + i;
            }
        return sum;
    }
    static void countDigits(int number) 
    {
        int count = 0;
        while (number > 0)
            {
                number = number / 10;
                count++;
            }
        System.out.println(count);
    }
}
